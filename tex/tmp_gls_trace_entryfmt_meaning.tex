\makeatletter
\AtBeginDocument{%
  \let\MTorig@gls@link\@gls@link
  \long\def\@gls@link[#1]#2#3{%
    \ifx#3\@glo@text
      \begingroup
      \edef\glslabel{\glsdetoklabel{#2}}%
      \ifx\glstype\undefined\protected@edef\glstype{\csname glo@\glslabel @type\endcsname}\fi
      \expandafter\ifx\csname gls@\glstype @entryfmt\endcsname\relax
        \typeout{MT-ENTRYFMT-NOT-FOUND for [\glstype]}%
      \else
        \expandafter\let\expandafter\MTentryfmt\csname gls@\glstype @entryfmt\endcsname
        \typeout{MT-ENTRYFMT-MEANING:[\detokenize\expandafter{\meaning\MTentryfmt}]}%
      \fi
      \endgroup
    \fi
    \MTorig@gls@link[#1]{#2}{#3}%
  }%
}
