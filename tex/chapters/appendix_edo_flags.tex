\section{EDO Flag Officers (snapshot)}\label{app:edo-flags}
\emph{\footnotesize Verify names/billets the week of your board. Current as of: 05 Dec 2025 (sourced from Navy flag officer biographies).}

\paragraph{Non-EDO billets (for awareness only).} The NAVSEA December~2025 org chart shows several key billets led by non-\ac{edo} officers: \ac{navsea}~08 (ADM William J.\ Houston, Submarine officer), \ac{peoiws} (RDML Thomas J.\ Dickinson, Surface Warfare AP), \ac{peomlb} (RDML Kevin P.\ Biehn, Surface Warfare AP), NAVSEA~05 (RDML Jay A.\ Seif, Submarine AP), \ac{peouws} (RDML Douglas J.\ Adams), \ac{peossbn}/\ac{sea}~07 (RDML Todd Weeks), and \ac{peousc} (currently SES; RDML Kevin R.\ Smith was relieved in May~2025)\autocite{navsea-org-chart-2025}.

\subsubsection*{Vice Admirals}
\begin{description}[style=nextline, labelsep=0.5em, font=\bfseries]
  \item[\textbf{\ac{vadm} Seiko Okano.}]
        \ac{pmildep} to \ac{asnrdanda}; former \ac{comnavwar} and \ac{peo} \ac{c4i} / Project Overmatch lead\autocite{navy-okano-2025}.
\end{description}

\subsubsection*{Rear Admirals (Upper Half)}
\begin{description}[style=nextline, labelsep=0.5em, font=\bfseries]
  \item[\textbf{\ac{radm} Brian A.\ Metcalf.}]
        \ac{peoships}; delivers surface combatants, auxiliaries, and amphibious ships\autocite{navy-metcalf-2025}.
  \item[\textbf{\ac{radm} Casey J.\ Moton.}]
        \ac{peocvn}; leads \ac{cvn} new construction and in-service nuclear availability portfolios\autocite{navy-moton-2025}.
  \item[\textbf{\ac{radm} Jonathan E.\ Rucker.}]
        \ac{peossn}; oversees attack submarine acquisition and lifecycle support\autocite{navy-rucker-2025}.
  \item[\textbf{\ac{radm} J.\ Kurt Rothenhaus.}]
        \ac{comnavwar}; commands \ac{navwar} and its \acp{pmw} delivering \ac{c4i}/cyber capabilities\autocite{navy-rothenhaus-2025}.
  \item[\textbf{\ac{radm} Douglas L.\ Williams.}]
        Director for Test, \ac{mda}; synchronizes joint missile defense test campaigns\autocite{navy-williams-2025}.
\end{description}

\subsubsection*{Rear Admirals (Lower Half)}
\begin{description}[style=nextline, labelsep=0.5em, font=\bfseries]
  \item[\textbf{\ac{rdml} Peter D.\ Small.}]
        \ac{navsea} Chief Engineer and Deputy Commander for Naval Systems Engineering; Commander, \ac{nswc}/\ac{nuwc}\autocite{navy-small-2025}.
  \item[\textbf{\ac{rdml} Dan L.\ Lannamann.}]
        \ac{cnrmc}; oversees regional maintenance centers and depot availabilities\autocite{navy-lannamann-2025}.
  \item[\textbf{\ac{rdml} Dianna Wolfson.}]
        Director, Fleet Maintenance, \ac{usff}; integrates \acp{tycom} readiness and ship maintenance execution\autocite{navy-wolfson-2024}.
  \item[\textbf{\ac{rdml} Daniel W.\ Ettlich.}]
        Director, Fleet Maintenance (N43), \ac{pacflt}; leads west-coast and forward-deployed maintenance readiness\autocite{navy-ettlich-2025}.
  \item[\textbf{\ac{rdml} Scott M.\ Brown.}]
        Deputy Commander, Industrial Operations, \ac{navsea}; oversees shipyard and industrial base sustainment\autocite{navy-brown-2025}.
  \item[\textbf{\ac{rdml} Huan Nguyen.}]
        Deputy Commander for Cyber Engineering, \ac{navsea}; integrates cybersecurity and digital engineering into ship programs\autocite{navy-nguyen-2021}.
  \item[\textbf{\ac{rdml} Jonathan J.\ Jett-Parmer.}]
        Deputy Commander, Supervisor of Shipbuilding, Conversion and Repair (\ac{navsea}); leads \ac{supship} technical authority for new construction and overhauls\autocite{navy-jett-parmer-2025}.
\end{description}
