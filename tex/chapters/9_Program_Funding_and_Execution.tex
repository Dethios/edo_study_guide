\section{Program Funding and Execution}

\subsection{Program Funding}
\subsubsection{Financial terms}
\begin{description}
  \item[\textbf{Budget authority.}] \ac{ba} is authority granted by appropriation law to enter obligations that will result in immediate or future outlays.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Commitment.}] Administrative reservation of funds (usually by the local comptroller) in anticipation of a future obligation.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Obligation.}] Legal reservation of funds to make a future payment of money.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Expenditure.}] Charge against available funds resulting from a voucher, claim, or other document approved by competent authority.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Outlay.}] Occurs when the vendor cashes the expenditure check and money flows from the Treasury to the vendor or supplier.\autocite{edo-3-1-3-program-funding-2025}
\end{description}

\begin{description}
  \item[\textbf{Credit-card analogy.}] \ac{ba} is the line of credit, commitment is the intent to purchase, obligation is the charge slip, expenditure is the bill/check, and outlay is the cash transfer from Treasury.\autocite{edo-3-1-3-program-funding-2025}
\end{description}

\subsubsection{Appropriation life-cycle and expired funds}
\begin{description}
  \item[\textbf{Current.}] Funds are available for new obligations during the appropriation's obligation window (1--5 years depending on category).\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Expired.}] A 5-year period where funds are no longer available for new obligations but may be used for obligation adjustments, expenditures, and outlays.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Canceled.}] Funds are closed; remaining balances return to Treasury.\autocite{edo-3-1-3-program-funding-2025}
\end{description}

\begin{description}
  \item[\textbf{Expired funds example 1.}] In Aug 2025, FY22 \ac{opn} funds are expired (FY22--FY24 obligation window), so a \ac{pco} cannot sign a new contract; the funds are limited to obligation adjustments, expenditures, and outlays.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Expired funds example 2.}] In Jan 2025, FY24 \ac{rdte} funds are still current (FY24--FY25), so a \ac{pco} may sign a new contract for work starting in FY25.\autocite{edo-3-1-3-program-funding-2025}
\end{description}

\note{Obligation windows: \ac{milpers} and \ac{omn} are 1 year; \ac{rdte} is 2 years; procurement is 3 years (\ac{scn} is 5); \ac{milcon} is 5 years.\autocite{edo-3-1-3-program-funding-2025}.}

\subsubsection{Funding policies}
\begin{description}
  \item[\textbf{Annual funding.}] Applies to expense appropriations (\ac{omn}, \ac{milpers}); request \ac{ba} only for the goods or services needed in a given fiscal year, with a major exception for non-severable service contracts that cross fiscal years.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Incremental funding.}] Applies to \ac{rdte}; request funding only for work expected to be performed in the given fiscal year. The Navy is authorized to use incremental funding for \ac{scn} on certain ship programs (\ac{cvn}, \ac{ssbn}, \ac{lha}/\ac{lhd}).\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Full funding.}] Applies to procurement and \ac{milcon}; annual budget requests must cover the total cost of complete, usable end items delivered within a 12-month period. If multiple end items are bought, do not budget for more items than can be delivered within 12 months; options can extend procurement across years.\autocite{edo-3-1-3-program-funding-2025}
\end{description}

\note{Incremental funding example: A 28-month, \$260M \ac{rdte} effort with FY24--FY26 costs of 35\%/45\%/20\% would request \$91M (FY24), \$117M (FY25), and \$52M (FY26).\autocite{edo-3-1-3-program-funding-2025}.}

\note{Full funding practical: A \ac{wpn} contract with deliveries accepted over two years violates full funding; fix by structuring lots or options so each lot delivers within 12 months and is funded with the year-of-delivery appropriation.\autocite{edo-3-1-3-program-funding-2025}.}

\subsubsection{Exceptions to full funding}
\begin{description}
  \item[\textbf{Advance procurement (long lead items).}] An exception to full funding for long-lead components; documented in the acquisition strategy, approved at milestone, and funded as a separate line item to protect schedules or critical skills.\autocite{edo-3-1-3-program-funding-2025}
\end{description}

\begin{description}
  \item[\textbf{Multiyear procurement (\ac{myp}).}] A single contract for a specific quantity of usable end items delivered over more than one year (no more than five). Congress authorizes the full contract scope at award (NDAA), but funding and appropriations remain annual.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Benefits.}] Stability in acquisition and budgeting, productivity gains, cost growth reduction, \ac{eoq} opportunities, and reluctance to cancel once in place.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Challenges.}] Termination liability if funding is not provided, inability to budget for termination liability, and political controversy (Congress and \ac{osd}).\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Required criteria.}] Savings relative to annual contracting (guidance ~10\%), stable requirements/design, confidence in cost estimates and production capability, and commitment to funding by Service, \ac{osd}, and Congress.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Initiation.}] Proposed in the President's Budget; if approved, authorized by the NDAA and funded through annual appropriations on a year-by-year basis.\autocite{edo-3-1-3-program-funding-2025}
\end{description}

\note{DDG-51 \ac{myp} example: The FY98--FY01 and FY02--FY05 \ac{myp} contracts saved over \$1B; the FY13--FY17 \ac{myp} saved about \$1.5B and enabled a 10th ship option.\autocite{edo-3-1-3-program-funding-2025}.}

\begin{description}
  \item[\textbf{Other exceptions.}] \ac{bb} (fewer restrictions than \ac{myp}), cost-to-complete (prior-year completion), and incremental funding for \ac{rcoh}.\autocite{edo-3-1-3-program-funding-2025}
\end{description}

\subsubsection{Funding product improvements}
\begin{description}
  \item[\textbf{Phase 1: Development and testing.}] \ac{rdte} when the modification enhances performance or requires independent \ac{dt}/\ac{iote}; otherwise Procurement for in-production systems, or \ac{omn} if the system is no longer in production.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Phase 2: Fabrication.}] Procurement.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Phase 3: Installation.}] Procurement.\autocite{edo-3-1-3-program-funding-2025}
\end{description}

\subsubsection{MILCON funding}
\begin{description}
  \item[\textbf{Program.}] The acquisition budget culminates in the annual Military Construction Program submitted to Congress for authorization and appropriation.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Threshold.}] \ac{milcon} is the acquisition (design and/or construction) of facilities costing more than \$4,000,000 in support of active and reserve requirements.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Sustainment.}] Maintenance and repair activities funded with \ac{omn}.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Modernization.}] Alteration or replacement of facilities to implement new standards, new functions, or components with 50+ year service life; funded with OPN.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Rule.}] It is expressly forbidden to incrementally fund a project to avoid \ac{milcon} triggers.\autocite{edo-3-1-3-program-funding-2025}
\end{description}

\subsubsection{Summary check}
\begin{description}
  \item[\textbf{Policies.}] Annual (\ac{omn}, \ac{milpers}); Incremental (\ac{rdte}, select \ac{scn}); Full Funding (Procurement, \ac{milcon}).\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Full funding exceptions.}] Advance Procurement and \ac{myp} are the primary exceptions; \ac{bb}, cost-to-complete, and \ac{rcoh} are additional cases.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Product improvements.}] Phase 1 uses \ac{rdte}/Procurement/\ac{omn} depending on testing and production status; Phases 2--3 use Procurement.\autocite{edo-3-1-3-program-funding-2025}
  \item[\textbf{Expired funds rule.}] Expired funds cannot create new obligations; they are limited to obligation adjustments, expenditures, and outlays.\autocite{edo-3-1-3-program-funding-2025}
\end{description}
