\section{NAVWAR Enterprise}

\subsection{Team NAVWAR overview}
\begin{description}
  \item[\textbf{Organization.}] The \ac{navwar} enterprise consists of the \ac{navwar} systems command (Echelon~II), three affiliated \acp{peo} (\ac{peoc4i}, \ac{peodigital}, \ac{peomlb}), and three Echelon~III activities: \ac{niwclant}, \ac{niwcpac}, and the \ac{nsfa}~\autocite{edo-2-1-3-navwar-enterprise-2024}.
  \item[\textbf{Team NAVWAR locations.}] More than 11{,}000 employees support fleet concentration areas and deployed operations worldwide~\autocite{edo-2-1-3-navwar-enterprise-2024}.
\end{description}

\subsection{Mission}
\begin{description}
  \item[\textbf{Mission statement.}] Identify, develop, deliver, and sustain information warfighting capabilities and services that enable naval, joint, coalition, and other national missions from seabed to space, while providing research and development, systems engineering, test and evaluation, technical, in-service, and support services to \acp{peo} across program life cycles~\autocite{edo-2-1-3-navwar-enterprise-2024}.
\end{description}

\subsection{Fleet Readiness Directorate (FRD)}
\begin{description}
  \item[\textbf{Fleet single point of entry.}] External engagement with \acp{tycom}, Fleet commanders, \acp{syscom}, regional maintenance centers, naval shipyards, \ac{navwar} directorates, and external \acp{pmo}; leads installation, fleet support, and cyber readiness~\autocite{edo-2-1-3-navwar-enterprise-2024}.
  \item[\textbf{NAVWAR accountability hub.}] Internal coordination with \acp{niwc}, \ac{navwar} directorates, and \acp{pmw} (100s, 700s)~\autocite{edo-2-1-3-navwar-enterprise-2024}.
  \item[\textbf{Execution rigor.}] Contracts with industry for afloat/ashore support, coordinates software loading and testing in \ac{cno} availabilities, and synchronizes hardware/software repair, patching, and sustainment~\autocite{edo-2-1-3-navwar-enterprise-2024}.
  \item[\textbf{Branches.}] FRD-100 (Fleet Support) and FRD-200 (Fleet Installations)~\autocite{edo-2-1-3-navwar-enterprise-2024}.
\end{description}

\subsection{Echelon III activities and core competencies}
\subsubsection{NIWC Pacific}
\begin{description}
  \item[\textbf{Mission.}] Conduct research, development, prototyping, engineering, test and evaluation, installation, and sustainment of integrated \ac{c4isr}, cyber, and space systems across all warfighting domains~\autocite{edo-2-1-3-navwar-enterprise-2024}.
  \item[\textbf{Technology areas.}] Cyber operations; command and control communications and networks; production, installation, and in-service engineering; intelligence, surveillance, reconnaissance and information operations; science and technology; marine mammal programs; autonomous unmanned systems~\autocite{edo-2-1-3-navwar-enterprise-2024}.
  \item[\textbf{Key capability areas.}] Full life-cycle engineering for information warfare; cloud-based digital development with \ac{devsecops} and digital twins; rapid prototyping and delivery; \ac{ai}/data science; applied electromagnetics and photonics; offensive/defensive cyber; positioning, navigation, and timing labs; and unique test facilities including San Clemente Island and integrated data-link test beds~\autocite{edo-2-1-3-navwar-enterprise-2024}.
\end{description}

\subsubsection{NIWC Atlantic}
\begin{description}
  \item[\textbf{Mission.}] Deliver information warfare solutions that protect national security, including communications systems, networking, cyber operations, intelligence and decision support applications, business systems, and information security~\autocite{edo-2-1-3-navwar-enterprise-2024}.
  \item[\textbf{Five departments.}] Fleet C4I and readiness; expeditionary warfare; enterprise systems; shore C2ISR and integration; science and technology~\autocite{edo-2-1-3-navwar-enterprise-2024}.
  \item[\textbf{Key capability areas.}] Surface ship and submarine C4I test and integration, cyber forensics and vulnerability assessment, enterprise data centers, satellite communications production and installation, electromagnetic interference testing, global distance support, air traffic control systems, and electronic security systems~\autocite{edo-2-1-3-navwar-enterprise-2024}.
\end{description}

\subsubsection{Naval Space Force Activity (NSFA)}
\begin{description}
  \item[\textbf{Mission.}] Lead the \ac{us} Navy's participation in the \ac{nro} by providing naval warfare and acquisition expertise, coordinating naval space \ac{rdte} with national reconnaissance programs, and enabling fleet use of national space capabilities~\autocite{edo-2-1-3-navwar-enterprise-2024}.
  \item[\textbf{Responsibilities.}] Facilitate interaction between the \ac{don} and \ac{nro}; coordinate research, development, and acquisition activities that support space-based intelligence collection~\autocite{edo-2-1-3-navwar-enterprise-2024}.
\end{description}

\subsection{Affiliated program executive offices}
\begin{description}
  \item[\textbf{PEO MLB.}] \ac{don} manpower, logistics, and business \ac{it} acquisition agent; delivers MyNavy HR IT, Navy ERP financials, logistics IT, applications services, data transformation, and enterprise systems/innovation support portfolios~\autocite{edo-2-1-3-navwar-enterprise-2024}.
  \item[\textbf{PEO Digital.}] \ac{don} enterprise-wide \ac{it} acquisition agent; portfolios include naval enterprise networks, special networks and intelligence mission applications, naval commercial cloud services, special access program networks, enterprise \ac{it} strategic sourcing, and Marine Corps supporting establishment systems~\autocite{edo-2-1-3-navwar-enterprise-2024}.
  \item[\textbf{PEO C4I.}] Delivers command and control, intelligence sharing, terminals, and integration programs through capability development offices (PMW~120/130/150/160/170) and platform integration offices (PMW~740/750/760/770/790) to provide affordable, integrated, and interoperable \ac{c4i} capability to the fleet~\autocite{edo-2-1-3-navwar-enterprise-2024}.
\end{description}
