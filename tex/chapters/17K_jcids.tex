\section{Joint Capabilities Integration \& Development System}

\subsection{Summary}
\begin{description}
  \item[\textbf{Decision support triad.}] \ac{jcids} (requirements), \ac{ppbe} (funding), and the \ac{das} (materiel) operate continuously and must stay aligned to enable timely decisions~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Purpose.}] \ac{jcids} ensures the joint warfighter receives required capabilities, institutionalizes joint needs, and initiates the acquisition management process~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{CBA foundation.}] \acp{cba} identify capability gaps and recommend materiel or non-materiel solutions based on operational risk and feasibility~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Capability documents.}] \ac{icd} and \ac{cdd} (with \ac{cdd} updates) align to acquisition phases and provide validated requirements for program execution~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Key players.}] \ac{cjcs}, \ac{jroc}, \ac{jcb}, \acp{fcb}, the Joint Staff Gatekeeper (J-8), and Joint Staff J-6 drive validation, prioritization, and interoperability decisions~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{Decision Support Systems}
\begin{description}
  \item[\textbf{Requirements.}] \ac{jcids} captures warfighter capability needs and requirements~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Funding.}] \ac{ppbe} provides the resource decisions to execute validated priorities~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Materiel.}] The \ac{das} executes acquisition strategies that deliver validated capabilities~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Synchronization.}] These systems must interface continuously to keep requirements, funding, and execution aligned~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{Capabilities-Based Assessment (CBA)}
\begin{description}
  \item[\textbf{Objective.}] A \ac{cba} analyzes mission needs and capability gaps, assesses risk, and recommends materiel and non-materiel solutions~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Initiation.}] A \ac{cba} can start from any \ac{dod} organization, a joint concept, a \ac{conops}, or an operational need~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Outputs.}] The \ac{cba} identifies mission and capability needs, gaps and risks, validates non-materiel feasibility, and recommends a path forward~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{CBA Phases}
\begin{description}
  \item[\textbf{Study definition.}] What does the mission require~\autocite{edo-3-3-3-jcids-2025}?
  \item[\textbf{Needs assessment.}] How well do current capabilities meet those needs; where are the gaps~\autocite{edo-3-3-3-jcids-2025}?
  \item[\textbf{Solutions recommendation.}] What materiel or non-materiel options should close the gaps~\autocite{edo-3-3-3-jcids-2025}?
\end{description}

\subsection{CBA Recommended Solutions}
\begin{description}
  \item[\textbf{Materiel.}] New or modified systems acquired through the \ac{das}; guided by validated \ac{jcids} documents~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Non-materiel.}] Changes in doctrine, organization, training, leadership, personnel, facilities, or policy; often faster and less costly than new systems~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Document path.}] Non-materiel solutions lead to a \ac{dcr}; materiel solutions lead to an \ac{icd}~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{DOTMLPF-P Elements}
\begin{description}
  \item[\textbf{Doctrine.}] Fundamental guiding principles~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Organization.}] How the \ac{dod} organizes to fight~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Training.}] How the force prepares to fight tactically~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Materiel.}] Existing equipment and systems~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Leadership and education.}] Leader development requirements~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Personnel.}] People required to accomplish the mission~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Facilities.}] Property, installations, and industrial facilities supporting forces~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Policy.}] Policies that impact military operations~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{Capability Documents}
\begin{description}
  \item[\textbf{Purpose.}] Capability documents communicate operator needs to acquisition, test, and resourcing communities~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Primary documents.}] \ac{icd} (approved prior to \ac{mdd}) and \ac{cdd} (approved prior to Milestone~B); \ac{cdd} updates support production~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{AoA support.}] \ac{aoa} is performed during the \ac{msa} phase as part of the requirements-to-acquisition handoff~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{Initial Capabilities Document (ICD)}
\begin{description}
  \item[\textbf{Purpose.}] Summarizes the \ac{cba} and justifies requirements to resolve capability gaps within a timeframe, including possible solutions for \ac{msa} and \ac{tmrr}~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Inputs.}] Summarizes \ac{dotmlpfp} analysis and capability gaps~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Supports.}] \ac{mdd}, \ac{aoa}, Acquisition Strategy, and Milestone~A~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{Capability Development Document (CDD)}
\begin{description}
  \item[\textbf{Purpose.}] Defines authoritative, measurable, and testable capabilities for \ac{emd} and the program technical baseline~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Affordability.}] Captures information needed to deliver an affordable and supportable capability across increments~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Performance.}] Specifies performance attributes, including \acp{kpp}, needed to design systems that close the gaps in the \ac{icd}~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Updates.}] Can be updated near end of development to refine production attributes, quantities, and subsequent increments~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{CDD Update and Legacy CPD}
\begin{description}
  \item[\textbf{Policy change.}] The 2018 \ac{jcids} manual removed the requirement for a separate \ac{cpd}; a \ac{cdd} update supports production needs~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Legacy CPD intent.}] Defined capabilities for Production and Deployment, refined \ac{cdd} thresholds/objectives, and was approved before Milestone~C after \ac{cdr}~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{JCIDS Within the Acquisition Framework}
\begin{figure}[htb]
  \centering
  \includegraphics[width=0.9\linewidth]{JCIDSinDAS.png}
  \caption[JCIDS Within the DAS]{Placement of \ac{jcids} documents within \ac{mca} phases. \srcCite{edo-3-3-3-jcids-2025}.}
  \label{fig:jcids-within-das}
\end{figure}

\begin{description}
  \item[\textbf{ICD.}] Defines capability gaps and supports \ac{mdd}, \ac{msa}, and \ac{tmrr} (no \acp{kpp})~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{CDD.}] Introduces \acp{kpp}, provides cost estimates, and guides \ac{emd} with measurable, testable capability definitions~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{CDD update.}] Refines \acp{kpp} and performance attributes, updates cost estimates, and guides Production and Deployment~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Team effort.}] Requirements development must involve the warfighter and all stakeholders to produce executable technical requirements~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{Key Players}
\begin{description}
  \item[\textbf{Gatekeeper (J-8/DDR).}] Manages document flow, assigns \ac{jsd}, and designates lead/supporting \acp{fcb} for staffing and validation~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Chairman of the Joint Chiefs of Staff.}] The \ac{cjcs} is the principal military adviser to the President, \ac{secdef}, and the \ac{nsc}; sets the agenda and presides over the Joint Chiefs of Staff~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Joint Requirements Oversight Council.}] The \ac{jroc} is the process owner; validates requirements, prioritizes joint capabilities, and validates \acp{kpp} unless delegated~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Joint Capabilities Board.}] The \ac{jcb} supports the \ac{jroc}, reviews documents, adjudicates issues, and validates \acp{kpp} for \ac{jcb}-interest documents~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Functional Capabilities Boards.}] The \acp{fcb} are permanent bodies that analyze, prioritize, and integrate stakeholder views across functional areas; include \ac{ccmd} participation~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{Joint Staff J-6 (C4/Cyber)}
\begin{description}
  \item[\textbf{Net-ready validation.}] Validates the Net-Ready performance attribute (\ac{kpp} or \ac{ksa}) for \ac{jroc}/\ac{jcb}-interest documents~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Interoperability certification.}] Performs IT and national security system interoperability and supportability certifications; \ac{jitc} issues certification before full-rate production~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{Importance of Interoperability}
\begin{description}
  \item[\textbf{Definition.}] Ability of systems, units, or forces to provide and accept services and operate effectively with U.S. and coalition partners~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Operational risk.}] Poor interoperability increases risk to people and equipment and reduces operational effectiveness~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Cost discipline.}] Stovepiped systems are costly and hinder joint operations; interoperability is a core \ac{jcids} validation priority~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{IT Acquisition Requirements}
\begin{description}
  \item[\textbf{Net-Ready attribute.}] Documented as a \ac{kpp} or \ac{ksa} to ensure IT supports operations, is manageable on networks, and exchanges information effectively~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Architecture compliance.}] Systems must align to \ac{dodaf}; architecture views are mandatory parts of the \ac{cdd} and must be consistent with \ac{disr} standards~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{Validation Authority}
\begin{description}
  \item[\textbf{Joint Requirements Oversight Council.}] The \ac{jroc} validates all ACAT~I and IA programs~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Joint Capabilities Board.}] The \ac{jcb} validates other ACAT~II/III programs with significant joint or interoperability impact~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Component sponsor.}] Component or sponsor validates all other requirements documents~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{Document Review and Staffing}
\begin{description}
  \item[\textbf{Submission.}] Sponsors submit documents to Joint Staff J-8 for review and staffing~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{JSD assignment.}] J-8 determines the document's joint impact and assigns a \ac{jsd} that drives staffing level and validation authority~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{FCB review.}] Lead \ac{fcb} reviews staffing results and recommends validation action~\autocite{edo-3-3-3-jcids-2025}.
  \item[\textbf{Validation.}] Confirms operational need, verifies non-materiel infeasibility, and ensures threat and certification requirements are met~\autocite{edo-3-3-3-jcids-2025}.
\end{description}

\subsection{Quick Review}
\begin{itemize}
  \item What system institutionalizes a joint capability need and sponsors further evaluation of a materiel solution~\autocite{edo-3-3-3-jcids-2025}? \textbf{Answer:} \ac{jcids}.
  \item Which acquisition phases use the \ac{icd} and \ac{cdd}~\autocite{edo-3-3-3-jcids-2025}? \textbf{Answer:} \ac{msa}/\ac{tmrr}, \ac{emd}, and Production and Deployment.
  \item Which document defines capabilities and is transformed into a program start~\autocite{edo-3-3-3-jcids-2025}? \textbf{Answer:} \ac{cdd}, with Milestone~B approval by the \ac{mdauthority}.
  \item JROC and the Joint Staff ensure new capabilities are conceived and developed in what context~\autocite{edo-3-3-3-jcids-2025}? \textbf{Answer:} Joint warfighting context.
  \item Warfighter gap identification is driven by Combatant Commander participation in what body~\autocite{edo-3-3-3-jcids-2025}? \textbf{Answer:} \acp{fcb}.
\end{itemize}
