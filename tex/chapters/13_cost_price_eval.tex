\section{Cost and Price Evaluation}

\subsection{Summary}
\begin{description}
  \item[\textbf{Fair and reasonable price.}] The \ac{ko} determines fair and reasonable price using price, cost, technical, field-pricing, and risk analyses along with sound business judgment~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Financial health signals.}] Balance sheets and income statements enable \ac{ros} and \ac{roa} calculations that inform responsibility and financial capacity decisions~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Direct v.s.\ indirect.}] Direct costs tie to a single cost objective; indirect costs (overhead and \ac{ganda}) support multiple objectives and must be allocated through defined bases~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Allowability tests.}] Billable costs must be reasonable, allocable, compliant with \ac{cas}/\ac{gaap}, within contract terms, and within \ac{far} 31.205 limitations~\autocite[\S~31.201-2]{FAR}~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Rate discipline.}] Indirect rates, fully burdened rates, and forward pricing rate agreements sharpen negotiation positions and protect the prenegotiation objective~\autocite[\S~15.407-3]{FAR}~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{description}

\subsection{Fair and Reasonable Price Determination}
\begin{description}
  \item[\textbf{KO responsibility.}] The \ac{ko} evaluates price reasonableness using price, cost, technical, field-pricing support, and risk analysis inputs to reach a fair and reasonable determination~\autocite[\S~15.404-1(a)]{FAR}~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Analysis toolkit.}] Price and cost analysis are primary techniques, supported by technical evaluations, field pricing support, and risk analysis as needed~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{description}

\subsection{Cost or Pricing Data Requirements}
\begin{description}
  \item[\textbf{Threshold and exceptions.}] The coursebook threshold is \$2M; certified cost or pricing data are generally required above the threshold unless a \ac{far} exception applies~\autocite[\S~15.403-1]{FAR}~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Solicitation request.}] When required, the Government must request cost or pricing data in the solicitation and typically seeks a technical analysis of labor, material, and other proposal elements~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{description}

\subsection{Cost and Price Analysis}
\begin{description}
  \item[\textbf{Price analysis.}] Examination of the total proposed price without evaluating separate cost elements, using competition, historical prices, or market indicators~\autocite[\S~15.404-1(b)]{FAR}.
  \item[\textbf{Cost analysis.}] Evaluation of individual cost elements and profit/fee to determine how well proposed costs represent what the contract should cost under reasonable economy and efficiency~\autocite[\S~15.404-1(c)]{FAR}~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Cost v.s.\ price lens.}] Cost analysis looks at direct labor, indirect rates, materials, and \acp{odc}; price analysis looks at total proposed price with profit/fee embedded~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{description}

\subsection{Field Pricing Support}
\begin{description}
  \item[\textbf{Specialist support.}] Field pricing support includes audits and specialist assessments (accountants, auditors, cost/price analysts, negotiators, engineers, production specialists, and small and disadvantaged business utilization specialists) to assist the \ac{ko} in determining reasonableness~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{DCAA/DCMA thresholds.}] \ac{dcma} provides field assistance and \ac{dcaa} supports audits for fixed-price proposals over \$10M and cost-type proposals over \$100M per the coursebook~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{description}

\subsection{Contractor Financial Health}
\begin{description}
  \item[\textbf{Balance sheet and income statement.}] Balance sheets show financial condition at a point in time; income statements summarize revenues and expenses over a period and enable profitability ratios~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{ROS and ROA.}] \ac{ros} and \ac{roa} gauge profitability and efficiency; compare ratios to industry averages and investigate sustained shortfalls~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Responsibility risk.}] Debt, cash-on-hand, and cash-flow trends complement ratio checks to determine offeror responsibility~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{description}

\subsection{Direct and Indirect Costs}
\begin{description}
  \item[\textbf{Direct costs.}] Costs traceable to a single cost objective (direct labor, direct material, and \acp{odc}) with clear identification to a specific requirement~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Indirect costs.}] Costs that benefit more than one objective and are allocated through overhead and \ac{ganda} pools, such as management, legal, travel, utilities, and facility security~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{description}

\subsection{Indirect Cost Pools, Bases, and Rates}
\begin{description}
  \item[\textbf{Cost pools.}] Indirect cost pools group similar indirect costs (e.g., material handling, design support, building maintenance) that are allocated to cost objectives using a logical, causal-beneficial relationship~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Allocation bases.}] Bases include direct labor dollars or hours, material dollars, machine hours, or units produced and are used to compute indirect rates~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{description}

Apply the standard relationships from the coursebook when validating proposals; they are board favorites for quick-turn computations covering indirect pools, \ac{tci}, \ac{ganda}, and profitability metrics such as \ac{ros} and \ac{roa}~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\begin{align*}
  \text{Indirect Cost Rate}        & = \frac{\text{Indirect Cost Pool}}{\text{Allocation Base}}\label{eq:indirect_rate}                                                \\
  \text{TCI}                       & = \text{Direct Cost} + \text{Overhead Cost}\label{eq:tci}                                                                         \\
  \text{G\&A Cost}                 & = \text{G\&A Rate} \times \text{TCI}\label{eq:ganda_cost}                                                                         \\
  \text{Fully Burdened Labor Rate} & = \frac{\text{Direct Labor Cost} + \text{Indirect Costs} + \text{Profit/Fee}}{\text{Direct Labor Hours}}\label{eq:fully_burdened} \\
  \text{ROS}                       & = \frac{\text{Operating Profit}}{\text{Net Sales}}\label{eq:ros}                                                                  \\
  \text{ROA}                       & = \frac{\text{Net Income}}{\text{Total Assets}}\label{eq:roa}
\end{align*}

\subsection{Cost Allowability}
\begin{description}
  \item[\textbf{Five bases.}] Reasonableness, allocability, compliance with \ac{cas}/\ac{gaap}, contract terms, and \ac{far} 31.205 limitations govern allowability~\autocite[\S~31.201-2]{FAR}~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Unallowable examples.}] Bad debts, contributions/donations, entertainment, and alcoholic beverages are unallowable~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{description}
\note{Slides may refer to ``cash allowability''; \ac{far} terminology is \emph{cost} allowability.}

\subsection{Fully Burdened Rate}
\begin{description}
  \item[\textbf{Definition.}] Fully burdened rates combine direct labor, indirect costs, and profit/fee to support comparisons across skill mixes and estimate labor-heavy work packages~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Example.}] A contractor with \$1.0M direct labor, \$0.5M manufacturing overhead, and \$0.2M other costs across 50,000 labor hours has a fully burdened rate of \$34 per hour~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{description}

\subsection{Forward Pricing Rates}
\begin{description}
  \item[\textbf{Purpose.}] \acp{fpra} establish agreed indirect rates (typically one to three years) to speed negotiations and reduce repetitive audits by anchoring the prenegotiation objective~\autocite[\S~15.407-3]{FAR}~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item[\textbf{Process.}] Contractors submit a \ac{fprp}; the Government reviews and negotiates rates into an \ac{fpra}, and the \ac{aco} monitors performance against the agreed rates~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{description}

\subsection{Practitioner Steps}
\begin{enumerate}
  \item Confirm whether certified cost or pricing data are required and document the applicable exception if not; ensure the solicitation includes any required data requests~\autocite[\S~15.403-1]{FAR}~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item Execute price analysis first, then cost analysis when price analysis alone cannot support reasonableness; integrate technical analysis and field pricing support as needed~\autocite[\S~15.404-1]{FAR}~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item Validate direct and indirect cost elements, confirm allocation bases, and apply the five allowability tests to each significant cost element~\autocite[\S~31.201-2]{FAR}.
  \item Analyze financial health using balance sheet and income statement ratios and note any responsibility concerns in the prenegotiation objective~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
  \item Capture \ac{dcaa} and \ac{dcma} inputs, forward pricing rate agreements, and fully burdened rate calculations in the contract file to justify the final determination~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{enumerate}

\subsection{Checklist}
\begin{itemize}
  \item File price and cost analysis results (with technical inputs) showing how the fair-and-reasonable price determination was reached~\autocite[\S~15.404-1]{FAR}.
  \item Verify each significant cost element passes the five allowability tests and that allocation bases match current practice~\autocite[\S~31.201-2]{FAR}.
  \item Document any \ac{fpra}/\ac{fprp} used in the prenegotiation objective, or explain why legacy rates remain valid~\autocite[\S~15.407-3]{FAR}.
  \item Record \ac{ros}/\ac{roa} trends, cash flow, and \ac{dcaa}/\ac{dcma} inputs that influenced the responsibility determination~\autocite{edo-3-2-4-cost-price-evaluation-2025}.
\end{itemize}
