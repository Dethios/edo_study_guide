\section{Ethics and Acquisition}

\subsection{Summary}
\begin{description}
  \item[\textbf{Ethical leadership.}] Ethical conduct is foundational to mission success and is the most important component of naval leadership~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Program success.}] Successful acquisition is judged differently by stakeholders (program managers, contractors, users, and Congress), so ethical choices balance competing perspectives~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Decision models.}] Ethical models (Golden Rule, Kant, utilitarianism, altruism) and the principled decision-making model provide structured ways to resolve dilemmas~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Influences.}] Ethical decisions are shaped by personal factors, organizational pressures, and professional context; recognizing these influences reduces risk~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
\end{description}

\subsection{Ethics and Successful Leadership}
\begin{description}
  \item[\textbf{Why it matters.}] Personal conduct is the most significant threat to a leader's career, so leaders must develop and sustain their own character while developing the character of their teams~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Misconduct patterns.}] Senior official misconduct often involves personal/ethical violations, improper personnel actions, misuse of government resources, travel violations, or procurement/security issues~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Causal factors.}] Common drivers include failure to consider consequences, misjudged privacy, alcohol abuse, loss of command stature, poor risk controls, weak moral compass, and temptation amplified by position and stress~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
\end{description}

\subsection{Ethical Decision-Making Foundations}
\begin{description}
  \item[\textbf{Decision types.}] Simple decisions follow established rules, while complex or gray-area decisions require deeper ethical analysis and deliberation~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Influences.}] Instinct, religion, education, gender, life experience, hierarchy, mental health, laws and regulations, society, peers, family, professional ethos, job position, and physical condition all influence ethical choices~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
\end{description}

\subsection{Ethical Decision-Making Models}
\begin{description}
  \item[\textbf{Golden Rule.}] Treat others the way you want to be treated; its limitation is that others may not want the same treatment~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Kant's absolute moral philosophy.}] Ethical principles always take precedence, with a duty to tell the truth; it lacks a method to choose among competing ethical values~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Consequentialism/Utilitarianism.}] Weigh outcomes to maximize overall good; this model can be manipulated to justify ends-over-means thinking~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Altruism.}] Act to increase others' welfare; it struggles when stakeholders have conflicting interests and offers limited guidance on trade-offs~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
\end{description}

\subsection{Principled Decision-Making Model}
\begin{description}
  \item[\textbf{Six pillars of character.}] Trustworthiness, respect, responsibility, justice/fairness, caring, and civic virtue/citizenship take precedence over non-ethical values~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Decision tests.}] The Washington Post test (public scrutiny), generalization test (precedent for others), and mirror test (personal conscience) validate choices~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
\end{description}
\begin{enumerate}
  \item \textbf{Stop and think.} Create time and space before deciding~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item \textbf{Clarify goals.} Balance short-term and long-term aims~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item \textbf{Determine facts.} Identify what you know and what you need to know~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item \textbf{Develop options.} Generate options and consult trusted advisors~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item \textbf{Consider consequences.} Assess stakeholder impacts and the six pillars~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item \textbf{Choose.} Make the decision and frame it with the decision tests~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item \textbf{Monitor and modify.} Reassess outcomes and adjust if needed~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
\end{enumerate}

\subsection{Acquisition Program Success Perspectives}
\begin{description}
  \item[\textbf{Program manager view.}] Success means delivering required capability on time and within budget~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Contractor view.}] Success means profitability and positive cash flow~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{User view.}] Success means fielding an effective and available capability~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Congressional view.}] Success balances social, environmental, and defense needs while ensuring appropriated funds are used for their intended purpose~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
\end{description}

\subsection{Acquisition Ethical Dilemmas}
\begin{description}
  \item[\textbf{Budget insight.}] Informal requests for \ac{rdte} impact data (e.g., a 30\% cut) require caution, transparency, and alignment with ethics rules and program leadership~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item[\textbf{Program pressures.}] Acquisition dilemmas (e.g., cost/schedule/performance problems with a prime contractor and congressional oversight) should be resolved using the principled model and the six pillars of character~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
\end{description}

\subsection{Quick Review}
\begin{itemize}
  \item Ethical leadership is essential to mission success and career longevity~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item The principled decision-making model provides seven steps anchored in the six pillars of character~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item Ethical models (Golden Rule, Kant, utilitarianism, altruism) help diagnose dilemmas but each has limits~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
  \item Acquisition success criteria vary by stakeholder, so ethical choices must balance competing priorities~\autocite{edo-5-1-3-ethics-and-acquisition-2025}.
\end{itemize}
