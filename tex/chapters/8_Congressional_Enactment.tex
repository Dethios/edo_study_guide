\section{Congressional Enactment}

\subsection{Constitutional basis for appropriations}
\begin{description}
  \item[\textbf{Article I authority.}] Congress provides for the common defense and general welfare and controls the Treasury.
        \begin{description}
          \item[\textbf{Section~8.}] \textit{\quote ``The Congress shall have power to... provide for the Common Defense... and general welfare...''}
          \item[\textbf{Section~9.}] \textit{\quote ``No money shall be drawn from Treasury, but in consequence of Appropriations made by law...''}
        \end{description}
  \item[\textbf{Appropriations.}] All funding to operate the federal government is appropriated by Appropriations Acts.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsection{Key terms}
\subsubsection{What is a budget?}
\begin{description}
  \item[\textbf{Definition.}] A financial plan for accomplishing an organization's objectives.
  \item[\textbf{Management tool.}] An instrument of planning, decision making, and management control.
  \item[\textbf{Priorities.}] A statement of priorities and fiscal policy.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsubsection{Budget definitions}
\begin{description}
  \item[\textbf{Budget authority.}] Authority granted by appropriation law to enter obligations that will result in immediate or future outlays.
  \item[\textbf{Commitment.}] Administrative reservation of funds, usually by the local comptroller, in anticipation of a future obligation.
  \item[\textbf{Obligation.}] Legal reservation of funds to make a future payment of money.
  \item[\textbf{Expenditure.}] Charge against available funds resulting from a voucher, claim, or other document approved by competent authority.
  \item[\textbf{Outlay.}] Occurs when the vendor cashes the expenditure check and money flows from the Treasury to the vendor or supplier.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsubsection{Authorization versus appropriation}
\begin{description}
  \item[\textbf{Authorization.}] A statute providing \ac{dod} legal authority to do something (for example, procure a ship or conduct operations). It authorizes or modifies programs and sets spending ceilings, but does not provide funding; drafted by \ac{hasc}/\ac{sasc}.
  \item[\textbf{Appropriation.}] A statute providing \ac{ba} for federal agencies; drafted by \ac{hac}/\ac{sac}.
  \item[\textbf{Rule.}] Defense programs must have authorization to conduct activities and appropriated funding to pay for them.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsection{Congressional roles and responsibilities}
\subsubsection{Personal versus professional staff}
\begin{description}
  \item[\textbf{Personal staff.}] Employees of individual members of Congress; provide constituent support and policy expertise with limited \ac{dod} interaction.
  \item[\textbf{Professional staff members (PSMs).}] Committee staff who review the President's Budget down to the program element; the primary congressional interface with \ac{dod}; \acp{pm} engage directly with PSMs.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsubsection{Oversight and analysis}
\begin{description}
  \item[\textbf{GAO.}] Congress's chief investigator, auditor, and program evaluator.
  \item[\textbf{CBO.}] Supports Budget and Appropriations Committees; analyzes the President's Budget, revenues, and outlays.
  \item[\textbf{CRS.}] Provides confidential, impartial research and analysis for Congress.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsubsection{Key committees}
\begin{description}
  \item[\textbf{Budget Committees.}] \ac{hbc}/\ac{sbc} craft the annual Budget Resolution.
  \item[\textbf{Armed Services Committees.}] \ac{hasc}/\ac{sasc} craft the \ac{ndaa}.
  \item[\textbf{Appropriations Committees.}] \ac{hac}/\ac{sac} craft annual Appropriations Bills.
  \item[\textbf{Conferences.}] Joint House/Senate meetings reconcile differences; members appointed by Speaker and Senate Majority Leader.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsection{Congressional enactment process}
\begin{description}
  \item[\textbf{Budget Resolution.}] Blueprint that sets top-line spending ceilings for 20 major government functions.
  \item[\textbf{Authorization Act.}] Approves programs, quantities, end strength, and policy (\ac{ndaa}).
  \item[\textbf{Appropriations Act.}] Provides \ac{ba} (legal authority to spend).
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsection{Budget Resolution}
\begin{description}
  \item[\textbf{Drafted by.}] \ac{hbc}/\ac{sbc}; process established by the Budget and Impoundment Control Act of 1974.
  \item[\textbf{Purpose.}] Sets spending ceilings for 20 major government functions; not a law and independent of the President's Budget.
  \item[\textbf{Process.}] CBO guidance; House and Senate develop and debate resolutions; conference resolves differences; concurrent budget resolution returned to floors.
  \item[\textbf{Goal.}] Budget Resolution by 15 April; appropriation sub-allocations follow the ceilings.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsection{Authorization}
\begin{description}
  \item[\textbf{NDAA.}] Drafted by \ac{hasc}/\ac{sasc}; allows programs to be established or continued; provides authorization for each \ac{dod} appropriation account.
  \item[\textbf{Annual authorization law.}] Program approval, procurement quantities, personnel end strength, funding ceilings, and policy updates.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsection{Appropriations}
\begin{description}
  \item[\textbf{Committees.}] \ac{hac}/\ac{sac}; 12 annual appropriations bills to operate the federal government.
  \item[\textbf{Subcommittees.}] Each bill has its own subcommittee; bills may be combined into consolidated or omnibus acts.
  \item[\textbf{Timing.}] Hearings kick off around March; committees should follow authorizer guidance.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsubsection{HAC/SAC subcommittees}
\begin{description}
  \item[\textbf{List.}] Agriculture, Rural Development, Food and Drug Administration; Commerce, Justice, Science; Defense; Energy and Water Development; Financial Services and General Government; Homeland Security; Interior and Environment; Labor, Health and Human Services, Education; Legislative Branch; Military Construction and Veterans Affairs; State and Foreign Operations; Transportation, Housing and Urban Development.
  \item[\textbf{Navy focus.}] Most Navy interaction is with the Defense subcommittees (HAC-D/SAC-D).
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsection{Appeals to congressional marks}
\begin{description}
  \item[\textbf{Definition.}] An appeal requests reconsideration of congressional adjustments (authorization or appropriation) to dollars, quantities, manpower, or language.
  \item[\textbf{Committee routing.}] Appeal to the next committee in the phase (for example, SASC, SAC, or the conference committee).
  \item[\textbf{Dollar rule.}] Appeal to the dollar amount closest to the President's Budget; generally to the higher of the two prior marks unless it exceeds the President's position.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsection{Continuing Resolution (CR)}
\begin{description}
  \item[\textbf{Definition.}] Special stopgap appropriations that keep government operating when annual appropriations are not enacted on time.
  \item[\textbf{Typical rules.}] Operate at current rates; new programs cannot start unless specified; funding limited to last year's levels or current committee marks.
  \item[\textbf{Impacts.}] Restrictive apportionment, limited obligation rates, and cost/schedule risk; shipbuilding may receive specific anomalies.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsection{Sequestration}
\begin{description}
  \item[\textbf{Definition.}] Automatic spending cuts through withdrawal of funding for certain government programs.
  \item[\textbf{Roles.}] CBO estimates statutory caps and sequestration risk; \ac{omb} determines if sequestration is required and its size.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}

\subsection{Summary check}
\begin{description}
  \item[\textbf{Three steps.}] Budget Resolution (top-line ceilings), Authorization (program approval and policy), Appropriations (\ac{ba}).
  \item[\textbf{Appropriations committees.}] Draft annual appropriations bills for the federal government.
  \item[\textbf{Continuing Resolution.}] Provides stopgap funding for the federal government.
\end{description}
~\autocite{edo-3-1-2-congressional-enactment-2025}
