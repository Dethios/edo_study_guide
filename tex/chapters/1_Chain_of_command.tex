\section{Command Chains and Governance}
\subsection{Operations Chain (Title 10)}

\begin{description}
  \item[\textbf{Chain flow.}] President $\to$ \ac{secwar} $\to$ \ac{ccdr}. The \ac{cjcs} is the principal military adviser; service chiefs are \emph{not} in the operational chain of Title 10.
  \item[\textbf{Board cue.}] Boards love the contrast with acquisition authority; avoid conflating operational and acquisition chains.
  \item[\textbf{Echelons.}] Counted from the \ac{cno} (e.g., \ac{cno} is Echelon I, \acp{syscom} are Echelon II, etc.).
\end{description}

Figure \ref{fig:chain_of_command_ops} shows the Title 10 chain of command. \Fig{\ref{fig:chain_of_command_org_chart}} shows the Chain of Command from the President to Echelon III commands.

\subsection{Acquisition Governance (Department of the Navy)}

\begin{description}
  \item[\textbf{Policy authority.}] \ac{asnrdanda} is the \ac{sae} for the \ac{don} and sets policy for \acp{peo} and \acp{drpm}~\autocite{SECNAV5400-15D,SECNAV5000-2G}.
  \item[\textbf{Matrix execution.}] \acp{peo} hold program authority; \acp{syscom} (\ac{navsea}, \ac{navwar}, \ac{navair}) provide the \ac{ta} warrants and workforce; \acp{nwc} deliver research, development, test, and engineering support~\autocite{SECNAV5400-15D}.
  \item[\textbf{Flag/PEO billets (current).}]
  \begin{itemize}
    \item[] \textbf{EDO-led today:} \ac{asnrdanda}'s \ac{pmildep} (\ac{vadm} Seiko Okano), \ac{comnavwar} (\ac{radm} J.\ Kurt Rothenhaus), \ac{peoships} (\ac{radm} Brian Metcalf), \ac{peocvn} (\ac{radm} Casey Moton), and \ac{peossn} (\ac{radm} Jonathan Rucker) anchor \ac{ta} and acquisition governance for the community~\autocite{navy-okano-2025,navy-rothenhaus-2025, navy-metcalf-2025,navy-moton-2025,navy-rucker-2025}.
    \item[] \textbf{Non-\ac{edo} leaders (awareness):} \ac{navsea}~08 is currently ADM William J.\ Houston (Submarine officer), \ac{peoiws} is RDML Thomas J.\ Dickinson (Surface Warfare Acquisition Professional), \ac{peomlb} is RDML Kevin P.\ Biehn (Surface Warfare Acquisition Professional), NAVSEA~05 is RDML Jay A.\ Seif (Submarine Acquisition Professional), and \ac{peousc} is filled by an SES (formerly RDML Kevin R.\ Smith)~\autocite{navsea-org-chart-2025}.
  \end{itemize}
  \FlagBilletNote{See Appendix~\ref{app:edo-flags} for the current flag billet roster and non-\ac{edo} billets.}
\end{description}
