\AtBeginDocument{%
  \let\MTorig@gls@link\@gls@link
  \long\def\@gls@link[#1]#2#3{%
    \ifx#3\@glo@text
      \begingroup
      \edef\glslabel{\glsdetoklabel{#2}}%
      \ifx\glstype\undefined\protected@edef\glstype{\csname glo@\glslabel @type\endcsname}\fi
      \expandafter\let\expandafter\MTfmt\csname gls@\glstype @entryfmt\endcsname
      \ifx\MTfmt\glsentryfmt
        \typeout{MT-ENTRYFMT-IS-GLSENTRYFMT}%
      \else
        \typeout{MT-ENTRYFMT-NOT-GLSENTRYFMT}%
      \fi
      \typeout{MT-ENTRYFMT-MEANING-RAW:[\meaning\MTfmt]}%
      \endgroup
    \fi
    \MTorig@gls@link[#1]{#2}{#3}%
  }%
}
