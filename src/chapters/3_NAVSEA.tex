\subsection{HQ Codes (must-know one-liners)}

\begin{description}
  \item[\textbf{\ac{sea}~01 -- Comptroller}] provides financial policy, advice and quality services to ensure \ac{navsea}'s customers' budgets are efficiently and effectively executed. \ac{sea} 01 manages appropriation areas as well as providing cost engineering, and industrial analysis.
  \item[\textbf{\ac{sea}~02 -- Contracts}] and its field contracting offices under the Contracts Competency award approximately \$36 billion in contracts annually for new construction ships and submarines, ship repair, major weapon systems and services.
  \item[\textbf{\ac{sea}~03 -- Cyber Engineering \& Digital Transformation}] has the responsibility for providing the infrastructure and support services for the programs and developers in the \ac{navsea} community. \ac{sea}~03 delivers combat power to the fleet through enterprise digital capabilities, infrastructure for cyber-secure digital work and innovation, and enhanced enterprise user's experience. \FlagBilletNote{\ac{rdml} Huan Nguyen serves as \ac{navsea} Deputy Commander for Cyber Engineering}
  \begin{itemize}
    \item Cyber Security: \ac{sea}~03 provides the \ac{navsea} Enterprise with guidance on cyber issues and ensures that all ships are able and ready to detect, defend and recover from cybersecurity attacks.
    \item \ac{it} Services: \ac{sea}~03 maintains and facilitates Network Operations and \ac{it} Service Delivery for Headquarters \ac{navsea} end users to enable maximum productivity for \ac{navsea} civilians and military personnel in the support of the warfighter.
    \item Digital Transformation: \ac{sea}~03 transforms \ac{navsea} into a digital capability adopting new digital technologies, capitalizing on data, increasing digital skills, and modernizing business processes.
  \end{itemize}
  \item[\textbf{\ac{sea}~04 -- Industrial Operations}] has the important mission of getting ships to sea and keeping them ready. \ac{sea}~04 is the preferred integrator of maintenance and industrial operations for its Enterprise customers. \ac{sea}~04 manages the four Naval Shipyards and the four \acp{supship}~\FlagBilletNote{\ac{rdml} Scott M.\ Brown is the \ac{navsea} Deputy Commander for Industrial Operations}
  \item[\textbf{\ac{sea}~05 -- Naval Systems Engineering \& Logistics Directorate (\ac{ta})}] is responsible for providing the engineering and scientific expertise, knowledge, and technical authority necessary to design, build, maintain, repair, modernize, certify, and dispose of the Navy's ships, submarines, and associated warfare systems. \ac{sea}~05 is organized into 16 groups. \FlagBilletNote{\ac{rdml} Peter D.\ Small is the \ac{navsea} Chief Engineer and Deputy Commander for Naval Systems Engineering (also Commander, \ac{nswc}/\ac{nuwc})}
  \begin{itemize}
    \item \ac{sea05b}
    \item \ac{sea05c}
    \item \ac{sea05e}
    \item \ac{sea05h}
    \item \ac{sea05m}
    \item \ac{sea05n}
    \item \ac{sea05p}
    \item \ac{sea05s}
    \item \ac{sea05t}
    \item \ac{sea05u}
    \item \ac{sea05v}
    \item \ac{sea05w}
    \item \ac{sea05z}
  \end{itemize}
  \item[\textbf{\ac{sea}~06 -- Sustainment.}] Life-cycle product support.
  \item[\textbf{\ac{sea}~07 -- Undersea Warfare.}] Dual-hatted as \ac{peossbn}; provides a full spectrum of research, development, test and evaluation, \ac{hme} systems engineering and fleet support services to the in-service submarine and undersea forces. \ac{subtech} coordinates the development of technologies to fulfill undersea warfare capability requirements.
  \item[\textbf{\ac{sea}~08 -- Nuclear Propulsion.}] Manages all technical matters pertaining to naval reactors. Triple-hat duties:
        \begin{description}
          \item[\textbf{\ac{sea}~08 Deputy Commander.}] Leads nuclear propulsion activities within \ac{navsea}.
          \item[\textbf{Director, Naval Nuclear Propulsion (\ac{opnav}~00N).}] Serves as the Navy's principal nuclear propulsion authority.
          \item[\textbf{Deputy Administrator for Naval Reactors.}] Interfaces with the National Nuclear Security Administration and the Department of Energy.
        \end{description}
  \item[\textbf{\ac{sea}~09 -- Safety and Regulatory Compliance.}] Strengthens and aligns safety oversight and reporting.
  \item[\textbf{\ac{sea}~10 -- Total Force and Corporate Operations}] performs all operations support for \ac{navsea} directorates and field activities as well as \acp{peo}.  Support includes administrative products and services, career planning, employee development, facilities, foreign military sales coordination, human resources, security, and university research assistance.
  \item[\textbf{\ac{sea}~21 -- Surface Ship Maintenance, Modernization and Sustainment}] manages the complete lifecycle support for all non-nuclear surface ships and is the principal interface with the Surface Warfare Enterprise. The directorate is responsible for the maintenance and modernization of non-nuclear surface ships currently operating in the Fleet. Through planned modernization and upgrade programs, \ac{sea}~21 will equip today's surface ships with the latest technologies and systems to keep them in the Fleet through their service lives. Additionally, \ac{sea}~21 oversees the ship inactivation process, including ship transfers or sales to friendly foreign navies, inactivation and/or disposal.
        \begin{description}
          \item[\textbf{\ac{pms}~321.}] Unmanned small combatants and amphibious ships.
          \item[\textbf{\ac{pms}~326.}] International fleet support.
          \item[\textbf{\ac{pms}~339.}] Surface training systems.
          \item[\textbf{\ac{pms}~421.}] Large surface combatant modernization and sustainment.
          \item[\textbf{\ac{pms}~443.}] Bridge integration and \ac{hme} sustainment.
          \item[\textbf{\ac{pms}~451.}] Destroyer Modernization~2.0 portfolio.
          \item[\textbf{\ac{sea}~21I.}] Inactive Ships Directorate.
          \item[\textbf{\ac{surfmepp} Activity.}] Surface Maintenance Engineering Planning Program.
        \end{description}
\end{description}
